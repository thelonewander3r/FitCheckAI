export async function createOccasion(intake: OccasionIntake): Promise<OccasionSession>
- store create (status fields not needed; simple session object with empty outfits/gaps).
- lookup venue: const venue = await getVenueProvider().lookupVenue({ venueName: intake.venueName, eventType: intake.eventType })
- load wardrobe items (from wardrobe-store listItems).
- compose: palette = merge venue.palette with skin-tone flattering colors if intake.skinTone is set (reuse the same mapping as context-engine: fair [navy,black,white,burgundy], light [navy,charcoal,sage,cream], medium [navy,olive,rust,cream], tan [black,white,gold,deep green], deep [white,emerald,gold,royal blue]; lowercase). palette = deduped lowercased union.
- const season = "any" (MVP; date-based season derivation is a later fast-follow).
- const composed = composeOutfits(items, { formality: formalityLevelToLabel(venue.formalityLevel), palette, season, presentation: intake.presentation ?? "neutral" })
- update store with venueContext, outfits: composed.outfits, gaps: composed.gaps, isMockMode: venue.isMock
- return the stored session.
